## FSx 実務Tips

## AWS CLI基本操作

### FSx for Windows File Server

```bash
# ファイルシステム作成
aws fsx create-file-system \
  --file-system-type WINDOWS \
  --storage-capacity 300 \
  --subnet-ids subnet-xxx \
  --security-group-ids sg-xxx \
  --windows-configuration \
    ThroughputCapacity=16,\
    ActiveDirectoryId=d-xxx,\
    AutomaticBackupRetentionDays=7

# ファイルシステム詳細取得
aws fsx describe-file-systems \
  --file-system-ids fs-xxx

# バックアップ作成
aws fsx create-backup \
  --file-system-id fs-xxx \
  --tags Key=Name,Value=manual-backup

# バックアップリスト
aws fsx describe-backups \
  --filters Name=file-system-id,Values=fs-xxx

# ファイルシステム削除
aws fsx delete-file-system \
  --file-system-id fs-xxx
```

### FSx for Lustre

```bash
# Lustreファイルシステム作成（Scratch）
aws fsx create-file-system \
  --file-system-type LUSTRE \
  --storage-capacity 1200 \
  --subnet-ids subnet-xxx \
  --lustre-configuration \
    DeploymentType=SCRATCH_2,\
    ImportPath=s3://my-bucket/data/,\
    ExportPath=s3://my-bucket/results/

# Persistentタイプ作成
aws fsx create-file-system \
  --file-system-type LUSTRE \
  --storage-capacity 1200 \
  --subnet-ids subnet-xxx \
  --security-group-ids sg-xxx \
  --lustre-configuration \
    DeploymentType=PERSISTENT_2,\
    PerUnitStorageThroughput=125,\
    AutomaticBackupRetentionDays=7

# データリポジトリタスク（S3同期）
aws fsx create-data-repository-task \
  --file-system-id fs-xxx \
  --type EXPORT_TO_REPOSITORY \
  --paths /

# タスクステータス確認
aws fsx describe-data-repository-tasks \
  --task-ids task-xxx
```

### FSx for NetApp ONTAP

```bash
# ONTAPファイルシステム作成
aws fsx create-file-system \
  --file-system-type ONTAP \
  --storage-capacity 1024 \
  --subnet-ids subnet-xxx subnet-yyy \
  --ontap-configuration \
    DeploymentType=MULTI_AZ_1,\
    ThroughputCapacity=256,\
    PreferredSubnetId=subnet-xxx

# Storage Virtual Machine (SVM) 作成
aws fsx create-storage-virtual-machine \
  --file-system-id fs-xxx \
  --name svm01 \
  --svm-admin-password MyPassword123

# ボリューム作成
aws fsx create-volume \
  --volume-type ONTAP \
  --name vol01 \
  --ontap-configuration \
    StorageVirtualMachineId=svm-xxx,\
    SizeInMegabytes=102400,\
    SecurityStyle=UNIX,\
    JunctionPath=/vol01

# スナップショット作成
aws fsx create-snapshot \
  --volume-id fsvol-xxx \
  --name snapshot-$(date +%Y%m%d)
```

### FSx for OpenZFS

```bash
# OpenZFSファイルシステム作成
aws fsx create-file-system \
  --file-system-type OPENZFS \
  --storage-capacity 64 \
  --subnet-ids subnet-xxx \
  --security-group-ids sg-xxx \
  --open-zfs-configuration \
    DeploymentType=SINGLE_AZ_1,\
    ThroughputCapacity=64,\
    RootVolumeConfiguration={DataCompressionType=ZSTD}

# 子ボリューム作成
aws fsx create-volume \
  --volume-type OPENZFS \
  --name project-a \
  --open-zfs-configuration \
    ParentVolumeId=fsvol-xxx,\
    StorageCapacityQuotaGiB=100,\
    DataCompressionType=ZSTD

# スナップショット作成
aws fsx create-snapshot \
  --volume-id fsvol-xxx \
  --name snapshot-$(date +%Y%m%d-%H%M%S)
```

## セットアップ

### FSx for Windows: クライアント接続

```powershell
# Windows PowerShell

# DNSネームの取得
$FileSystemDNSName = "fs-xxx.example.com"

# ネットワークドライブマッピング
New-PSDrive -Name "Z" `
  -PSProvider FileSystem `
  -Root "\\$FileSystemDNSName\share" `
  -Persist

# または
net use Z: \\fs-xxx.example.com\share /persistent:yes

# 接続テスト
Test-Path Z:\

# ACL設定例
$Acl = Get-Acl "Z:\folder"
$Permission = "DOMAIN\User","FullControl","ContainerInherit,ObjectInherit","None","Allow"
$AccessRule = New-Object System.Security.AccessControl.FileSystemAccessRule $Permission
$Acl.SetAccessRule($AccessRule)
Set-Acl "Z:\folder" $Acl
```

### FSx for Lustre: Linux マウント

```bash
# Amazon Linux 2

# Lustreクライアントインストール
sudo amazon-linux-extras install -y lustre

# マウントポイント作成
sudo mkdir -p /mnt/fsx

# マウント
sudo mount -t lustre \
  fs-xxx.fsx.ap-northeast-1.amazonaws.com@tcp:/fsx \
  /mnt/fsx

# 永続化 (/etc/fstab)
echo "fs-xxx.fsx.ap-northeast-1.amazonaws.com@tcp:/fsx /mnt/fsx lustre defaults,noatime,flock,_netdev 0 0" | \
  sudo tee -a /etc/fstab

# パフォーマンステスト
time dd if=/dev/zero of=/mnt/fsx/testfile bs=1M count=1024

# S3同期確認
ls -lh /mnt/fsx/
```

### FSx for NetApp ONTAP: NFSマウント

```bash
# Ubuntu/Debian

# NFS クライアントインストール
sudo apt-get update
sudo apt-get install -y nfs-common

# マウント（NFS v4.1）
sudo mkdir -p /mnt/ontap

sudo mount -t nfs \
  -o nfsvers=4.1,rsize=1048576,wsize=1048576,timeo=600,retrans=2 \
  svm-xxx.fs-xxx.fsx.ap-northeast-1.amazonaws.com:/vol01 \
  /mnt/ontap

# 永続化
echo "svm-xxx.fs-xxx.fsx.ap-northeast-1.amazonaws.com:/vol01 /mnt/ontap nfs4 nfsvers=4.1,rsize=1048576,wsize=1048576,timeo=600,retrans=2,_netdev 0 0" | \
  sudo tee -a /etc/fstab
```

### トラブルシューティング

```bash

ping fs-xxx.fsx.ap-northeast-1.amazonaws.com
telnet fs-xxx.fsx.ap-northeast-1.amazonaws.com 988

# CloudWatch メトリクス確認
aws cloudwatch get-metric-statistics \
  --namespace AWS/FSx \
  --metric-name DataReadBytes \
  --dimensions Name=FileSystemId,Value=fs-xxx \
  --start-time 2026-07-26T00:00:00Z \
  --end-time 2026-07-26T23:59:59Z \
  --period 3600 \
  --statistics Average,Maximum

# ログ確認
# CloudWatch Logs
aws logs tail /aws/fsx/windows/fs-xxx --follow
```
